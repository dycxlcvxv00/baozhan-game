function makeTex(scene,key,w,h,draw){const g=scene.make.graphics({x:0,y:0,add:false});draw(g,w,h);g.generateTexture(key,w,h);g.destroy();}
function glowCircle(g,cx,cy,r,color,a1){for(let i=3;i>=1;i--){g.fillStyle(color,(a1/i)*0.5);g.fillCircle(cx,cy,r*i/1.6);}g.fillStyle(color,a1);g.fillCircle(cx,cy,r*0.55);}

  // 暗黑氛围背景
