function castBolt(scene){const t=nearest();if(!t)return;burst(scene,t.x,t.y,'bolt',0xfff36b,14,200);let dmg=90;let cur=t,hitN=0;const hitSet=new Set();while(cur&&hitN<5){damageMonster(scene,cur,dmg,'lightning');hitSet.add(cur);dmg*=.85;hitN++;let nxt=null,bd=1e9;G.monsters.getChildren().forEach(m=>{if(m.hp>0&&!hitSet.has(m)){const d=Math.hypot(m.x-cur.x,m.y-cur.y);if(d<bd){bd=d;nxt=m;}}});if(nxt&&bd<260){burst(scene,nxt.x,nxt.y,'bolt',0xfff36b,8,160);cur=nxt;}else break;}scene.cameras.main.flash(60,255,240,150);}
function damageMonster(scene,m,dmg,elem){if(m.hp<=0)return;m.hp-=dmg;m.hit=.12;m.setTint(0xffffff);burst(scene,m.x,m.y,elem==='fire'?'fire':elem==='ice'?'ice':'arrow',elem==='lightning'?0xfff36b:0xffffff,4,100);if(m.hp<=0)killMonster(scene,m);}
function killMonster(scene,m){G.kills++;G.gold+=m.type==='boss'?200:m.type==='demon'?15:8;burst(scene,m.x,m.y,m.type==='boss'?'fire':'skel',m.type==='boss'?0xff3b30:0xcfc8bb,m.type==='boss'?30:10,m.type==='boss'?260:160);if(m.type==='boss')scene.cameras.main.shake(300,.01);if(m.hpBar)m.hpBar.destroy();m.destroy();}
function endBattle(scene,win){G.state=win?'win':'lose';const cx=W/2,cy=H/2;darkPanel(scene,cx,cy,440,240);scene.add.text(cx,cy-60,win?'胜  利':'石墙告破',{fontFamily:'serif',fontSize:'40px',color:win?'#ffd23f':'#ff5f64'}).setOrigin(.5).setShadow(0,0,win?'#ffd23f':'#ff3b30',18);scene.add.text(cx,cy+2,win?`恶魔狂潮已被击退\n金币 +${G.gold}`:'恶魔踏破了城墙',{fontFamily:'serif',fontSize:'16px',color:'#9a8ab8',align:'center'}).setOrigin(.5);const btn=scene.add.text(cx,cy+78,'再 战 一 场',{fontFamily:'serif',fontSize:'22px',color:'#1a0f24',backgroundColor:'#c9a6ff',padding:{x:28,y:12}}).setOrigin(.5).setInteractive({useHandCursor:true});btn.on('pointerdown',()=>{scene.scene.restart();});}

function update(time,delta){
  const dt=Math.min(.05,delta/1000);
  // 石墙血条
  G.wallBar.clear();G.wallBar.fillStyle(0x000000,.6).fillRect(8,H-26,220,14);G.wallBar.fillStyle(0xb03040,1).fillRect(8,H-26,220*Math.max(0,G.wall.hp/G.wall.max),14);G.wallBar.lineStyle(1,0xffd23f,.5).strokeRect(8,H-26,220,14);
  G.hud.wave.setText(G.state==='battle'?`第 ${G.wave} 波 / 共 ${G.waveTotal} 波`:'');
  G.hud.gold.setText(`金币 ${G.gold}`);G.hud.kills.setText(`击杀 ${G.kills}`);
  if(G.state!=='battle')return;
  G.time+=dt;
  // 出怪
  G.spawnT+=dt;
  for(const it of G.spawnQueue){if(!it.done&&G.spawnT>=it.at){spawnMonster(this,it.type);it.done=true;}}
  // 英雄自动普攻
  G.atkT=(G.atkT||0)-dt;if(G.atkT<=0){G.atkT=1.0;heroAttack(this);}
  // 技能充能自动释放
  G.charge.fire+=dt*8;G.charge.ice+=dt*8;G.charge.bolt+=dt*8;
  if(G.charge.fire>=COST.fire){G.charge.fire=0;castFire(this);}
  if(G.charge.ice>=COST.ice){G.charge.ice=0;castIce(this);}
  if(G.charge.bolt>=COST.bolt){G.charge.bolt=0;castBolt(this);}
  // 怪物推进与攻城
  G.monsters.getChildren().forEach(m=>{
    m.hit=Math.max(0,(m.hit||0)-dt);if(m.hit<=0)m.clearTint();
    m.slow=Math.max(0,(m.slow||0)-dt);
    if(m.x-m.r>G.wall.x){m.x-=m.speed*(m.slow>0?.5:1)*dt;}
    else{m.atkT-=dt;if(m.atkT<=0){m.atkT=1.4;G.wall.hp-=m.atk;this.cameras.main.shake(80,.004);burst(this,m.x,m.y,'fire',0xff5f64,4,90);if(G.wall.hp<=0)endBattle(this,false);}}
    if(m.barUpdate)m.barUpdate();
  });
  // 投射物
  G.bolts.getChildren().forEach(b=>{
    b.x+=b.vx*dt;b.y+=b.vy*dt;
    if(b.kind==='fireball'&&b.tx&&Math.hypot(b.x-b.tx,b.y-b.ty)<26){burst(this,b.x,b.y,'fire',[0xff7a2a,0xffd23f],20,240);G.monsters.getChildren().forEach(m=>{if(m.hp>0&&Math.hypot(m.x-b.x,m.y-b.y)<90)damageMonster(this,m,b.dmg,'fire');});b.destroy();return;}
    let dead=b.x>W+60||b.x<-60||b.y<-60||b.y>H+60;
    if(!dead){for(const m of G.monsters.getChildren()){if(m.hp>0&&!b.hit.has(m)&&Math.hypot(m.x-b.x,m.y-b.y)<m.r+8){b.hit.add(m);if(b.kind==='arrow')damageMonster(this,m,b.dmg,'arrow');else if(b.kind==='ice'){damageMonster(this,m,b.dmg,'ice');m.slow=3;}b.pierce--;if(b.pierce<=0){dead=true;break;}}}}
    if(dead)b.destroy();
  });
  // 胜负判定
  const allSpawned=G.spawnQueue.every(i=>i.done);
  if(allSpawned&&G.monsters.getChildren().length===0){if(G.wave>=G.waveTotal)endBattle(this,true);else nextWave(this);}
}
