function startBattle(scene){clearMenu();G.state='battle';G.wave=0;G.time=0;G.kills=0;G.wall.hp=G.wall.max;G.charge={fire:0,ice:0,bolt:0};nextWave(scene);}
function nextWave(scene){G.wave++;G.spawnQueue=buildWave(G.wave);G.spawnT=0;}
function buildWave(n){
  const q=[];const boss=(n===G.waveTotal);
  const base=n===1?6:n===2?9:12;
  for(let i=0;i<base;i++){q.push({type:i%4===2?'demon':'skel',at:i*Math.max(.7,1.3-n*.15)});}
  if(boss){q.push({type:'boss',at:base*.7+3});}
  return q.sort((a,b)=>a.at-b.at);
}
function spawnMonster(scene,type){
  const scale=1+(G.wave-1)*.15;
  const x=W+60,y=90+Math.random()*(H-180);
  const m=scene.add.image(x,y,type).setScale(type==='boss'?1.6:1);
  m.type=type;m.hp=m.maxHp=Math.round((type==='boss'?900:type==='demon'?180:80)*scale);
  m.speed=(type==='boss'?18:type==='demon'?30:52)*(1+ (G.wave-1)*.05);
  m.atk=type==='boss'?60:type==='demon'?30:16;m.atkT=0;m.r=type==='boss'?40:type==='demon'?20:16;m.slow=0;m.hit=0;
  m.hpBar=scene.add.graphics();
  scene.tweens.add({targets:m,y:y+(Math.random()<.5?-10:10),duration:1400+Math.random()*800,yoyo:true,repeat:-1,ease:'sine.inout'});
  m.barUpdate=()=>{m.hpBar.clear();if(m.hp<m.maxHp){m.hpBar.fillStyle(0x000000,.6).fillRect(m.x-m.r,m.y-m.r-12,m.r*2,5);m.hpBar.fillStyle(0x70e19b,1).fillRect(m.x-m.r,m.y-m.r-12,m.r*2*Math.max(0,m.hp/m.maxHp),5);}};
  if(type==='boss'){scene.cameras.main.shake(400,.012);scene.add.particles(x,y,'fire',{speed:{min:-120,max:120},lifespan:700,scale:{start:1.4,end:0},tint:[0xff3b30,0x8a1f6e],blendMode:'ADD',quantity:24,emitting:false}).explode(24,x,y);}
  G.monsters.add(m);
}
